
## 本地服务器部署
server.js:
```
const express = require('express')
const history = require('connect-history-api-fallback')

const PORT = 8088

const app = express()

// 插件，处理 前端 history 路由刷新404问题
app.use(history())

app.use(express.static(__dirname + '/public'))

app.listen(PORT, () => {
  console.log(`Server is running http://localhost:${PORT}`)
})
```
把打包后的文件放在 public 目录下，启动 npx nodemon server.js，打开
```
http://localhost:8088/
``` 
即可访问。

## nginx 服务器部署

安装 nginx， conf 文件夹下有 nginx.conf 配置文件，修改端口号为8099：
```
server {
    listen       8099;
    server_name  localhost;

    location / {
        root   html;
        index  index.html index.htm;
    }
    ...
}
```
双击 nginx 安装包根目录下的 nginx.exe, 打开 localhost:8099 及可访问到 nginx安装包/html/index.html 页面。

设置让 nginx 打开 前端项目打包后的页面，更改root html路径（或者直接替换/html/index.html），进程管理器里停止 nginx，再重启：
```
server {
    listen       8099;
    server_name  localhost;

    location / {
        root   E:\xhh\myproject\xhh-project\dist;
        index  index.html index.htm;
    }
    ...
}
```
处理 前端 history 路由刷新404问题，重启服务：
```
location / {
    root   E:\xhh\myproject\xhh-project\dist;
    index  index.html index.htm;
    try_files $uri $uri/ /index.html;
}
```

## 打包后自动上传服务器
```
const { execSync } = require('child_process')
const { SCP } = require('scp2')
const { NodeSSH } = require('node-ssh')
const compressing = require('compressing')
const chalk = require('chalk')
const path = require('path')
const fs = require('fs')

// 配置你的服务器信息
const config = {
  localDist: path.resolve(__dirname, './dist'), // 本地打包目录
  remotePath: '/www/wwwroot/xxx.com', // 服务器目录
  host: '1.2.3.4',
  port: 22,
  username: 'root',
  password: 'xxx',
  // privateKey: '~/.ssh/id_rsa', // 密钥登录（二选一）
  buildCommand: 'npm run build', // 打包命令
  autoBackup: true, // 是否自动备份
}

const ssh = new NodeSSH()
const log = (msg) => console.log(chalk.cyan(`[部署] ${msg}`))
const success = (msg) => console.log(chalk.green(`[成功] ${msg}`))
const error = (msg) => console.log(chalk.red(`[错误] ${msg}`))

// 部署流程
async function deploy() {
  try {
    log('1. 开始打包：' + config.buildCommand)
    execSync(config.buildCommand, { stdio: 'inherit' })
    success('打包完成')

    log('2. 压缩 dist 包')
    const zipFile = path.resolve(__dirname, 'dist.zip')
    await compressing.zip.compressDir(config.localDist, zipFile)
    success('压缩完成')

    log('3. 连接服务器')
    await ssh.connect({
      host: config.host,
      port: config.port,
      username: config.username,
      password: config.password,
      // privateKey: config.privateKey,
    })
    success('服务器连接成功')

    // 可选：备份线上目录
    if (config.autoBackup) {
      log('4. 自动备份线上文件')
      await ssh.execCommand(`zip -r ${config.remotePath}_backup_${Date.now()}.zip ${config.remotePath}`)
      success('备份完成')
    }

    log('5. 上传压缩包')
    await new Promise((resolve, reject) => {
      SCP.scp(
        zipFile,
        {
          host: config.host,
          port: config.port,
          username: config.username,
          password: config.password,
          path: config.remotePath,
        },
        (err) => (err ? reject(err) : resolve())
      )
    })
    success('上传完成')

    log('6. 服务器解压覆盖')
    await ssh.execCommand(`unzip -o ${config.remotePath}/dist.zip -d ${config.remotePath}`)
    success('解压完成')

    log('7. 清理临时文件')
    fs.unlinkSync(zipFile)
    await ssh.execCommand(`rm -f ${config.remotePath}/dist.zip`)
    success('清理完成')

    success('✅ 前端项目部署完成！')

    // 上传后刷新 Nginx
    await ssh.execCommand('nginx -s reload')

    process.exit(0)
  } catch (err) {
    error('部署失败：' + err.message)
    console.log(err)
    process.exit(1)
  }
}

// 执行
deploy()
```
